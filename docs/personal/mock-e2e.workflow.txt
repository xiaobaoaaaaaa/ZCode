name: Personal fork validation
on:
  pull_request:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  windows-source-build:
    if: github.repository == 'xiaobaoaaaaaa/ZCode'
    runs-on: windows-latest
    timeout-minutes: 60
    env:
      ZCODE_DATA_BASE_DIR: ${{ github.workspace }}/.personal-validation-home
      ZCODE_ENV: production
      DO_NOT_TRACK: "1"
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 10.33.2
      - uses: actions/setup-node@v4
        with:
          node-version: 24.14.0
      - name: Install locked dependencies without application startup
        run: $env:Pnpm_Max_Workers='1'; pnpm install --frozen-lockfile --ignore-scripts
      - name: Typecheck
        run: pnpm typecheck
      - name: Targeted presentation, memory and identity tests
        run: pnpm exec vitest run packages/ui/test/compactExecutionInteraction.test.ts packages/ui/test/v4ConversationTurnGroup.test.ts packages/ui/test/memorySettingsSection.test.ts packages/services/test/globalMemoryService.test.ts packages/desktop/test/desktopProductIdentity.test.ts
      - name: Compile desktop source (no installer, signing or publishing)
        run: pnpm --filter @zcode/desktop build:no-runtime-assets
      - name: Preserve compiled source bundles for review
        uses: actions/upload-artifact@v4
        with:
          name: personal-windows-source-bundles
          path: packages/desktop/out
          retention-days: 7
          if-no-files-found: error
      - name: Verify preview orchestration contracts
        run: node --test scripts/test/personal-windows-preview.test.mjs
      - name: Prepare unsigned Windows x64 unpacked preview
        run: node scripts/personal-windows-preview.mjs
      - name: Smoke packaged runtime and renderer in a temporary profile
        run: node scripts/personal-windows-smoke.mjs
      - name: Personal synthetic model protocol and persistence suite
        run: node scripts/personal-mock-suite.mjs
      - name: Personal synthetic model Electron UI regression
        run: node scripts/personal-mock-renderer.mjs
      - name: Archive unpacked candidate and checksum
        shell: pwsh
        run: |
          Add-Type -AssemblyName System.IO.Compression.FileSystem
          $output = Join-Path $env:GITHUB_WORKSPACE 'packages/desktop/dist-personal-preview'
          $zip = Join-Path $output 'ZCode-Personal-windows-x64-preview.zip'
          [System.IO.Compression.ZipFile]::CreateFromDirectory((Join-Path $output 'win-unpacked'), $zip)
          $digest = (Get-FileHash -Algorithm SHA256 $zip).Hash.ToLowerInvariant()
          "$digest  ZCode-Personal-windows-x64-preview.zip" | Set-Content (Join-Path $output 'SHA256SUMS.txt')
      - name: Upload unsigned preview candidate only after successful smoke
        uses: actions/upload-artifact@v4
        with:
          name: personal-windows-x64-preview
          path: |
            packages/desktop/dist-personal-preview/*.zip
            packages/desktop/dist-personal-preview/SHA256SUMS.txt
            packages/desktop/dist-personal-preview/preview-manifest.json
          retention-days: 7
          if-no-files-found: error
      - name: Preserve preview smoke evidence including failures
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: personal-windows-preview-diagnostics
          path: packages/desktop/dist-personal-preview/smoke-evidence
          retention-days: 7
          if-no-files-found: warn

      - name: Preserve synthetic model evidence including failures
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: personal-mock-e2e-evidence
          path: |
            packages/desktop/dist-personal-preview/mock-evidence/**
            packages/desktop/dist-personal-preview/mock-protocol-evidence/**
          if-no-files-found: ignore
          retention-days: 7
