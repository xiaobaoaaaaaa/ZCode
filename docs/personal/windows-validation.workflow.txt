name: Personal fork validation
on:
  pull_request:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  windows-source-build:
    if: github.repository == 'xiaobaoaaaaaa/ZCode'
    runs-on: windows-latest
    timeout-minutes: 30
    env:
      ZCODE_DATA_BASE_DIR: ${{ github.workspace }}/.personal-validation-home
      ZCODE_ENV: production
      DO_NOT_TRACK: "1"
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 10.33.2
      - uses: actions/setup-node@v4
        with:
          node-version: 24.14.0
      - name: Install locked dependencies without application startup
        run: pnpm install --frozen-lockfile --ignore-scripts
      - name: Typecheck
        run: pnpm typecheck
      - name: Targeted presentation, memory and identity tests
        run: pnpm exec vitest run packages/ui/test/compactExecutionInteraction.test.ts packages/ui/test/v4ConversationTurnGroup.test.ts packages/ui/test/memorySettingsSection.test.ts packages/services/test/globalMemoryService.test.ts packages/desktop/test/desktopProductIdentity.test.ts
      - name: Compile desktop source (no installer, signing or publishing)
        run: pnpm --filter @zcode/desktop build:no-runtime-assets
      - name: Preserve compiled source bundles for review
        uses: actions/upload-artifact@v4
        with:
          name: personal-windows-source-bundles
          path: packages/desktop/out
          retention-days: 7
          if-no-files-found: error
